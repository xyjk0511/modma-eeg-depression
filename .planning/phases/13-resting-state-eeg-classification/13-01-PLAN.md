# Plan 13-01: Resting-State EEG Classification

## Objective
Extract Welch PSD band-power features from MODMA resting-state EEG and run LOSO classification to compare with ERP BA=0.670.

## Tasks

### Task 1: Add `load_resting_features()` to `run_modma_erp.py`
**File**: `run_modma_erp.py`
**Action**: Add constant `RESTING_DIR` and function after the existing config block.

```python
RESTING_DIR = Path("854301_EEG_128Channels_Resting_Lanzhou_2015/EEG_128channels_resting_lanzhou_2015")
BANDS = {"delta":(1,4),"theta":(4,8),"alpha":(8,13),"beta":(13,30),"gamma":(30,45)}

def load_resting_features():
    from scipy.io import loadmat
    from scipy.signal import welch
    X, y, ids = [], [], []
    for f in sorted(RESTING_DIR.glob("*.mat")):
        sid = f.stem
        mat = loadmat(f)
        data = mat["data"][:128, :]          # (128, T), drop ref ch 128
        fs = float(mat["samplingRate"].flat[0])
        freqs, psd = welch(data, fs=fs, nperseg=256)  # (128, F)
        feats = []
        for lo, hi in BANDS.values():
            mask = (freqs >= lo) & (freqs < hi)
            feats.append(psd[:, mask].mean(axis=1))   # (128,)
        X.append(np.concatenate(feats))               # (640,)
        label = 1 if sid.startswith("0201") else 0
        y.append(label); ids.append(sid)
    return np.array(X), np.array(y), ids
```

### Task 2: Add `run_resting_analysis()` to `run_modma_erp.py`
**File**: `run_modma_erp.py`
**Action**: Add function after `load_resting_features()`.

```python
def run_resting_analysis():
    out_dir = Path("out_phase13"); out_dir.mkdir(exist_ok=True)
    X, y, ids = load_resting_features()
    n = len(y)
    pipe = Pipeline([
        ("scaler", StandardScaler()),
        ("pca",    PCA(n_components=min(20, n - 1))),
        ("clf",    LogisticRegression(C=1.0, class_weight="balanced", max_iter=1000)),
    ])
    loo = LeaveOneOut()
    preds, probs = np.zeros(n), np.zeros(n)
    for tr, te in loo.split(X):
        pipe.fit(X[tr], y[tr])
        preds[te] = pipe.predict(X[te])
        probs[te]  = pipe.predict_proba(X[te])[:, 1]
    ba  = balanced_accuracy_score(y, preds)
    auc = roc_auc_score(y, probs)
    lines = [
        f"Resting-state EEG classification (N={n})",
        f"BA={ba:.3f}  AUC={auc:.3f}",
        f"ERP baseline: BA=0.670",
        f"Delta BA: {ba - 0.670:+.3f}",
    ]
    print("\n".join(lines))
    (out_dir / "resting_result.txt").write_text("\n".join(lines))
```
